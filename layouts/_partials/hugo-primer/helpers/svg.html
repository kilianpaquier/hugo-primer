{{- /* Renders an inline SVG fetched at build time, with an optional tooltip. Fails the build when the SVG can't be read. @param {string} [class] The CSS classes of the SVG, its base name is always appended. @param {string} [height] The SVG height. @param {string} src The SVG source, a local path or a remote URL starting with https://. @param {string} [tooltip] The tooltip text, also used as accessible label. @param {string} [width] The SVG width. @returns {template.HTML} @examples {{ partial "hugo-primer/helpers/svg.html" (dict "src" "/static/primer/tag-16.svg" "width" "16") }} {{ partial "hugo-primer/helpers/svg.html" (dict "src" "https://cdn.simpleicons.org/gitlab" "width" "22" "tooltip" "GitLab") }} */}} {{- $src := .src }} {{- $remote := hasPrefix $src "https://" }} {{- $resource := dict }} {{- if $remote }} {{- with try (resources.GetRemote $src) }} {{- with .Err }} {{- errorf "%s" . }} {{- else with .Value }} {{- $resource = .Content }} {{- else }} {{- errorf "Unable to get remote resource %q" $src }} {{- end }} {{- end }} {{- else }} {{- with try ($src | readFile) }} {{- with .Err }} {{- errorf "%s" . }} {{- else with .Value }} {{- $resource = . }} {{- else }} {{- errorf "Unable to get resource %q" $src }} {{- end }} {{- end }} {{- end }} {{- $base := path.BaseName $src }} {{- $oldsvg := "" }} {{- $attrs := slice }} {{- if .class }} {{- $attrs = $attrs | append (printf "class=%q" (print .class " " $base)) }} {{- else }} {{- $attrs = $attrs | append (printf "class=%q" $base) }} {{- end }} {{- if .height }}{{ $attrs = $attrs | append (printf "height=%q" .height) }}{{ end }} {{- if .width }}{{ $attrs = $attrs | append (printf "width=%q" .width) }}{{ end }} {{- if .tooltip }} {{- $id := replace (.tooltip | lower) " " "-" }} {{- $attrs = $attrs | append (printf "id=%q" $id) }} {{- $attrs = $attrs | append (printf "aria-labelledby=%q" (print "tooltip-" $id)) }} {{ .tooltip }} {{- end }} {{- $str := "" }} {{- range $attrs }}{{ $str = print $str " " . }}{{ end }} {{- $newsvg := printf `` $str }} {{ replaceRE $oldsvg $newsvg $resource | safeHTML }}
