{{- /* Bundles remote and local stylesheets into one minified and fingerprinted link element. Local files are executed as templates with the site as context, and SCSS files are transpiled with site.Params.hugo_primer.styles.transpiler. Empty entries are skipped and nothing is rendered when no file remains. @param {slice} [local] The asset paths of local CSS or SCSS files. @param {slice} [remote] The URLs of remote CSS files. @returns {template.HTML} @examples {{ partialCached "hugo-primer/helpers/styles.html" (dict "local" (slice "styles/main.scss")) "main" }} */}} {{- $merged := slice }} {{- range .remote }} {{- if eq . "" }}{{ continue }}{{ end }} {{- with resources.GetRemote . }} {{- $merged = $merged | append . }} {{- end }} {{- end }} {{- range .local }} {{- $url := . }} {{- if eq $url "" }}{{ continue }}{{ end }} {{- $scss := hasSuffix . ".scss" }} {{- $build := printf "css/%s-%s-build.css" (path.BaseName $url) site.Language.Lang }} {{- $tmpl := printf "%s/%s-%s-tmpl.%s" (cond $scss "scss" "css") (path.BaseName $url) site.Language.Lang (cond $scss "scss" "css") }} {{- $opts := dict "enableSourceMap" (not hugo.IsProduction) "outputStyle" (cond hugo.IsProduction "compressed" "expanded") "targetPath" $build "transpiler" site.Params.hugo_primer.styles.transpiler }} {{- with resources.Get . }} {{- with resources.ExecuteAsTemplate $tmpl site . }} {{- $merged = $merged | append (cond $scss (. | toCSS $opts) .) }} {{- end }} {{- end }} {{- end }} {{- if gt (len $merged) 0 }} {{- with $merged | resources.Concat (path.Join "css" (printf "%s.css" (md5 math.Rand))) | minify | fingerprint }} {{- end }} {{- end }}
